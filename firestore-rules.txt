rules_version = '2';

//  打字勇者（index.html）Firestore 安全規則
//
//  貼上位置：Firebase 主控台 → Firestore Database → 規則 → 全部取代 → 發布
//  專案：school-4c9bf
//
//  重要：這份規則同時保留了舊站 index (2).html 用的 competition / scores，
//        直接整份取代即可，舊站不會壞掉。
// ═══════════════════════════════════════════════════════════════

service cloud.firestore {
  match /databases/{database}/documents {

    // ── 舊站「班級打字挑戰賽」使用（維持原本行為，不要動）──
    match /competition/{doc} {
      allow read, write: if request.auth != null;
    }
    match /scores/{doc} {
      allow read, write: if request.auth != null;
    }

    // ── 打字勇者：全班共用設定（時間、難度、模式、班級清單）──
    match /mg_config/{doc} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null;
    }

    // ── 打字勇者：開戰狀態（教師按開始/結束時寫入）──
    match /mg_round/{doc} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null;
    }

    // ── 打字勇者：題庫（9 池，教師編輯後存回）──
    match /mg_bank/{doc} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null;
    }

    // ── 打字勇者：成績（文件 ID = 班級_座號，例如「五年級3班_12號」）──
    match /mg_scores/{doc} {
      allow read:   if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if request.auth != null;
    }
  }
}
