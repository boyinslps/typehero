rules_version = '2';

// ═══════════════════════════════════════════════════════════════
//  打字勇者 Firestore 安全規則：正式版（docs/SPEC-TEACHER-ADMIN.md §1.3）
//
//  貼上位置：Firebase 主控台 → Firestore Database → 規則 → 全部取代 → 發布
//  專案：school-4c9bf
//
//  ⚠ 上線步驟（老師在「沒有上課的時段」一次做完，三步隔越短越好）：
//    ① Authentication 開「電子郵件／密碼」→ 建老師帳號 → 複製 UID；
//       Firestore 手動建 mg_admins/{UID}，欄位 ok = true
//       （一定要在第②步之前，不然老師自己也被擋在外面）
//    ② 貼這份規則並發布。貼之前先複製舊規則存檔，出事能貼回去
//    ③ 同一時間，新程式（A 版）上 main
//    沒有過渡版：②到③之間，舊程式的序號、房間、回報、老師控制台會被擋，所以愈短愈好。
//    這段時間開著舊分頁的學生要按 F5 重新整理，才會載到新程式。
//
//  老師身分 = 在 mg_admins 有自己的 UID 文件。repo 裡不放 UID 與 email。
//
//  做不到的事：匿名登入的 uid 對不上「班級_座號」，所以擋不住學生改自己的
//  字元與裝備。這份規則擋的是：學生自己寄信、開活動、批准回報、改題庫、刪全校存檔。
//
//  舊站 competition / scores 維持原樣（舊站還在用）。
//  規則本身不看 mg_scores 的欄位，遊戲新增存檔欄位不用回來改。
// ═══════════════════════════════════════════════════════════════

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }
    // 老師 = mg_admins 裡有自己的 UID（只能從主控台建）
    function isTeacher() {
      return request.auth != null
        && exists(/databases/$(database)/documents/mg_admins/$(request.auth.uid));
    }

    // ── 舊站「班級打字挑戰賽」使用（不要動）──
    match /competition/{doc} {
      allow read, write: if signedIn();
    }
    match /scores/{doc} {
      allow read, write: if signedIn();
    }

    // ── 老師名單：只能讀自己那份，誰都不能從程式寫 ──
    match /mg_admins/{uid} {
      allow get: if request.auth != null && request.auth.uid == uid;
      allow list, write: if false;
    }

    // ── 設定（時間、難度、序號、通關密語…）：學生只能讀 ──
    match /mg_config/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    // 提名投票：學生只能動投票 v 與提名 noms，其他欄位（隱藏、清空）只有老師
    // 文件還不存在時學生第一次提名是 create，所以 create 也要限制欄位
    match /mg_config/nw_vote {
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['v', 'noms']);
      allow update: if signedIn()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['v', 'noms']);
    }

    // ── 開戰狀態、題庫：只有老師能寫 ──
    match /mg_round/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_bank/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }

    // ── 成績（文件 ID = 班級_座號）：刪除只有老師 ──
    match /mg_scores/{doc} {
      allow read: if signedIn();
      allow create, update: if signedIn();
      allow delete: if isTeacher();
    }

    // ── 組隊房間（文件 ID = 班級_座號）與成員 ──
    match /mg_rooms/{room} {
      allow read, write: if signedIn();
      match /members/{member} {
        allow read, write: if signedIn();
      }
    }

    // ── 排行的場次紀錄：學生只能新增，老師全部 ──
    match /mg_nw_runs/{run} {
      allow read: if signedIn();
      allow create: if signedIn();
      allow update, delete: if isTeacher();
    }

    // ── 回報、回饋、換名紀錄：裡面有學生真名，只有老師能讀 ──
    match /mg_nw_flags/{flag} {
      // 老師頒獎、搬移舊資料也要能 create，所以老師不受下面的欄位限制
      allow create: if isTeacher() || (request.auth != null
        && request.resource.data.type in ['bug', 'feedback', 'namechg']
        && request.resource.data.get('status', 'pending') == 'pending'
        && request.resource.data.keys().hasOnly(['type', 'status', 'docId', 'cat', 'text', 'did', 'what', 'view',
             'mk', 'mon', 'star', 'rate', 'cls', 'seat', 'name', 'oldName', 't'])
        && request.resource.data.get('text', '').size() <= 400
        && request.resource.data.get('did', '').size() <= 200
        && request.resource.data.get('what', '').size() <= 200);
      // 回饋送出會用同一個 ID 重試：第一次其實寫進去了只是逾時，第二次變成 update。
      // 內容完全一樣就放行。
      allow update: if isTeacher() || request.resource.data == resource.data;
      allow read, delete: if isTeacher();
    }

    // ── 序號領取紀錄（一人一份）──
    // 文件 ID = 雜湊_班級_座號。學生只能 create：同一人第二次 create 會變成 update，被擋＝不能重複領。
    match /mg_gift_used/{doc} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['h', 'id', 't'])
        && request.resource.data.h is string;
      allow update, delete: if isTeacher();
    }

    // ── 伺服器時間：算學生電腦時差用，只放 t ──
    match /mg_clock/{doc} {
      allow read: if signedIn();
      allow create, update: if signedIn()
        && request.resource.data.keys().hasOnly(['t']);
      allow delete: if isTeacher();
    }

    // ── 信箱、活動、佈告欄公告、抓蟲榜：學生只能讀（抓蟲榜裡不放真名，見下面 mg_bugboard_private）──
    match /mg_mail/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_events/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_news/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_bugboard/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    // ── 抓蟲榜的老師專用資料：手動加分的理由、老師核可的真名。學生連讀都不行 ──
    // mg_bugboard 所有登入的人都讀得到，所以老師打的字與真名只能放這裡
    match /mg_bugboard_private/{doc} {
      allow read, write: if isTeacher();
    }

    // ── 老師操作紀錄：只有老師能讀、只能新增 ──
    match /mg_admin_log/{doc} {
      allow read: if isTeacher();
      allow create: if isTeacher();
    }
  }
}
