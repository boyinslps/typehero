rules_version = '2';

// ═══════════════════════════════════════════════════════════════
//  打字勇者 Firestore 安全規則：正式版（docs/SPEC-TEACHER-ADMIN.md §1.3）
//
//  貼上位置：Firebase 主控台 → Firestore Database → 規則 → 全部取代 → 發布
//  專案：school-4c9bf
//
//  ⚠ 上線步驟（老師在「沒有上課的時段」一次做完，三步隔越短越好）：
//    ① Authentication 開「電子郵件／密碼」→ 建老師帳號 → 複製 UID；
//       Firestore 手動建 mg_admins/{UID}，欄位 ok = true
//       （一定要在第②步之前，不然老師自己也被擋在外面）
//    ② 貼這份規則並發布。貼之前先複製舊規則存檔，出事能貼回去
//    ③ 同一時間，新程式（A 版）上 main
//    沒有過渡版：②到③之間，舊程式的序號、房間、回報、老師控制台會被擋，所以愈短愈好。
//    這段時間開著舊分頁的學生要按 F5 重新整理，才會載到新程式。
//
//  老師身分 = 在 mg_admins 有自己的 UID 文件。repo 裡不放 UID 與 email。
//
//  做不到的事：匿名登入的 uid 對不上「班級_座號」，所以擋不住學生改自己的
//  字元與裝備。這份規則擋的是：學生自己寄信、開活動、批准回報、改題庫、刪全校存檔。
//
//  舊站 competition / scores 維持原樣（舊站還在用）。
//  規則本身不看 mg_scores 的欄位，遊戲新增存檔欄位不用回來改。
// ═══════════════════════════════════════════════════════════════

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }
    // 老師 = mg_admins 裡有自己的 UID（只能從主控台建）
    function isTeacher() {
      return request.auth != null
        && exists(/databases/$(database)/documents/mg_admins/$(request.auth.uid));
    }

    // ── 舊站「班級打字挑戰賽」使用（不要動）──
    match /competition/{doc} {
      allow read, write: if signedIn();
    }
    match /scores/{doc} {
      allow read, write: if signedIn();
    }

    // ── 老師名單：只能讀自己那份，誰都不能從程式寫 ──
    match /mg_admins/{uid} {
      allow get: if request.auth != null && request.auth.uid == uid;
      allow list, write: if false;
    }

    // ── 設定（時間、難度、序號、通關密語…）：學生只能讀 ──
    match /mg_config/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    // 提名投票：學生只能動投票 v 與提名 noms，其他欄位（隱藏、清空）只有老師
    // 文件還不存在時學生第一次提名是 create，所以 create 也要限制欄位
    match /mg_config/nw_vote {
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['v', 'noms']);
      allow update: if signedIn()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['v', 'noms']);
    }

    // ── 開戰狀態、題庫：只有老師能寫 ──
    match /mg_round/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_bank/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }

    // ── 成績（文件 ID = 班級_座號）：刪除只有老師 ──
    match /mg_scores/{doc} {
      allow read: if signedIn();
      allow create, update: if signedIn();
      allow delete: if isTeacher();
    }

    // ── 組隊房間（文件 ID = 班級_座號）與成員 ──
    match /mg_rooms/{room} {
      allow read, write: if signedIn();
      match /members/{member} {
        allow read, write: if signedIn();
      }
    }

    // ── 狩獵的場次紀錄（2b 開始寫，2a 榮譽碑才讀；docs/SPEC-STAGE2.md §2.6）──
    // 每場一筆，ID＝「文件ID_毫秒」。學生只能新增：
    //  · 不能 update／delete：不然可以把舊紀錄的 ts 刷成最近、灌進步榜。同一個 ID 重送會被擋（permission-denied），
    //    程式把它當成「已經寫進去了」。
    //  · 不能讀：每題的時間不必讓同學看到；榜單由老師端彙總成 mg_nw_board 給學生讀。
    //  · ts 必須是伺服器時間；分週、分「兩週以前／最近兩週」一律用它，改電腦時間塞不了舊紀錄。
    match /mg_nw_runs/{run} {
      allow read: if isTeacher();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['id', 'cls', 'seat', 'uid', 't', 'ts', 'v', 'app', 'mk', 'rk', 'star', 'lv', 'qb', 'res',
             'secs', 'acc', 'ans', 'cor', 'typ', 'keys', 'nokey', 'act', 'cpm', 'fl', 'sus', 'w', 'atk', 'pw', 'tier', 'q',
             'frac', 'flow', 'guards', 'vines', 'dodgeReal', 'idle'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.ts == request.time
        && run.matches(request.resource.data.id + '_[0-9]+')
        // 8 分鐘（480 秒）魔物逃走的場次也要寫得進去，留 10 秒的誤差
        && request.resource.data.secs is number && request.resource.data.secs >= 0 && request.resource.data.secs <= 490
        // SPEC-BALANCE-V2 §7 第 4 步的遙測欄位：舊版程式不會送，所以全部可省略（get 給預設值）；有送就要在範圍內。
        // frac＝打掉的血比例；flow＝流派（代號字串，或流派傷害占比，兩種寫法都收，免得第 4 步定案後還要再貼一次規則）；
        // guards／vines／dodgeReal＝次數；idle＝沒在打字的秒數
        && request.resource.data.get('frac', 0) is number && request.resource.data.get('frac', 0) >= 0 && request.resource.data.get('frac', 0) <= 1
        && ((request.resource.data.get('flow', '') is string && request.resource.data.get('flow', '').size() <= 12)
            || (request.resource.data.get('flow', 0) is number && request.resource.data.get('flow', 0) >= 0 && request.resource.data.get('flow', 0) <= 1))
        && request.resource.data.get('guards', 0) is int && request.resource.data.get('guards', 0) >= 0 && request.resource.data.get('guards', 0) <= 999
        && request.resource.data.get('vines', 0) is int && request.resource.data.get('vines', 0) >= 0 && request.resource.data.get('vines', 0) <= 999
        && request.resource.data.get('dodgeReal', 0) is int && request.resource.data.get('dodgeReal', 0) >= 0 && request.resource.data.get('dodgeReal', 0) <= 999
        && request.resource.data.get('idle', 0) is number && request.resource.data.get('idle', 0) >= 0 && request.resource.data.get('idle', 0) <= 490
        && request.resource.data.acc is number && request.resource.data.acc >= 0 && request.resource.data.acc <= 1
        && request.resource.data.star is int && request.resource.data.star >= 1 && request.resource.data.star <= 5
        // q 是平的數字陣列，每 4 個一題 [字數, 按鍵, 毫秒, 打對]，最多 120 題（Firestore 不能存陣列裡的陣列）
        && request.resource.data.q is list && request.resource.data.q.size() <= 480 && request.resource.data.q.size() % 4 == 0
        && request.resource.data.sus is list && request.resource.data.sus.size() <= 4;
      allow update, delete: if isTeacher();
    }

    // ── 榮譽碑榜單（2a）：老師端彙總後寫，學生只讀；累積狀態與「先不上榜」名單只有老師看得到 ──
    match /mg_nw_board/{doc} { allow read: if signedIn(); allow write: if isTeacher(); }
    match /mg_nw_board_private/{doc} { allow read, write: if isTeacher(); }

    // ── 回報、回饋、換名紀錄：裡面有學生真名，只有老師能讀 ──
    match /mg_nw_flags/{flag} {
      // 老師頒獎、搬移舊資料也要能 create，所以老師不受下面的欄位限制
      allow create: if isTeacher() || (request.auth != null
        && request.resource.data.type in ['bug', 'feedback', 'namechg', 'run']
        && request.resource.data.get('status', 'pending') == 'pending'
        && request.resource.data.keys().hasOnly(['type', 'status', 'docId', 'cat', 'text', 'did', 'what', 'view',
             'mk', 'mon', 'star', 'rate', 'cls', 'seat', 'name', 'oldName', 't', 'runId', 'why'])
        && request.resource.data.get('runId', '').size() <= 80
        && request.resource.data.get('why', []).size() <= 4
        && request.resource.data.get('text', '').size() <= 400
        && request.resource.data.get('did', '').size() <= 200
        && request.resource.data.get('what', '').size() <= 200);
      // 回饋送出會用同一個 ID 重試：第一次其實寫進去了只是逾時，第二次變成 update。
      // 內容完全一樣就放行。
      allow update: if isTeacher() || request.resource.data == resource.data;
      allow read, delete: if isTeacher();
    }

    // ── 序號領取紀錄（一人一份）──
    // 文件 ID = 雜湊_班級_座號。學生只能 create：同一人第二次 create 會變成 update，被擋＝不能重複領。
    match /mg_gift_used/{doc} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['h', 'id', 't'])
        && request.resource.data.h is string;
      allow update, delete: if isTeacher();
    }

    // ── 伺服器時間：算學生電腦時差用，只放 t ──
    match /mg_clock/{doc} {
      allow read: if signedIn();
      allow create, update: if signedIn()
        && request.resource.data.keys().hasOnly(['t']);
      allow delete: if isTeacher();
    }

    // ── 信箱、活動、佈告欄公告、抓蟲榜：學生只能讀（抓蟲榜裡不放真名，見下面 mg_bugboard_private）──
    match /mg_mail/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_events/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_news/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    match /mg_bugboard/{doc} {
      allow read: if signedIn();
      allow write: if isTeacher();
    }
    // ── 抓蟲榜的老師專用資料：手動加分的理由、老師核可的真名。學生連讀都不行 ──
    // mg_bugboard 所有登入的人都讀得到，所以老師打的字與真名只能放這裡
    match /mg_bugboard_private/{doc} {
      allow read, write: if isTeacher();
    }

    // ── 老師操作紀錄：只有老師能讀、只能新增 ──
    match /mg_admin_log/{doc} {
      allow read: if isTeacher();
      allow create: if isTeacher();
    }
  }
}
